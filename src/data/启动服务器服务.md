Script ocr_book.sh:

```bash
#!/bin/bash
# usage: ocr_book.sh <pdf> <outdir> <dpi> <first_page> <last_page>
PDF="$1"; OUT="$2"; DPI="$3"; A="$4"; B="$5"
mkdir -p "$OUT/pages"
for ((p=A; p<=B; p++)); do
  txt="$OUT/pages/p$(printf %03d $p).txt"
  [ -s "$txt" ] && continue
  pdftoppm -f $p -l $p -r $DPI -gray -png "$PDF" "$OUT/tmp_$p" 2>/dev/null
  img=$(ls "$OUT"/tmp_$p-*.png 2>/dev/null | head -1)
  [ -z "$img" ] && continue
  OMP_THREAD_LIMIT=1 tesseract "$img" "${txt%.txt}" -l chi_sim --psm 6 2>/dev/null
  rm -f "$OUT"/tmp_$p-*.png
done
echo "DONE $OUT" >> "$OUT/done.flag"
```
